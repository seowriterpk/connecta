   429→function RowHeartButton({ id }: { id: string }) {

   453→function GroupRow({ group, searchQuery }: { group: GroupCardDTO; searchQuery: string }) {
   454→  const isActiveRecently = group.lastActiveAt
   455→    ? Date.now() - new Date(group.lastActiveAt).getTime() < 7 * 86400000
   456→    : false;
   457→  const isHot = group.members >= 1000;
   458→  const isTrending = group.views >= 50 || group.members >= 700;
   459→  const gradient = ROW_GRADIENTS[group.category?.color ?? "emerald"] ?? ROW_GRADIENTS.emerald;
   460→  // Adult directive: blank image title/alt on adult groups (SEO safety).
   461→  const imgAlt = group.isAdult ? "" : group.title;
   462→  const imgTitle = group.isAdult ? undefined : group.title;
   463→
   464→  return (
   465→    <motion.article
   466→      whileHover={{ y: -2 }}
   467→      transition={{ type: "spring", stiffness: 320, damping: 24 }}
   468→      className="group relative flex items-center gap-3 overflow-hidden rounded-2xl border bg-card p-3.5 pr-3 shadow-sm transition-[box-shadow,border-color] duration-300 hover:border-primary/30 hover:shadow-lg hover:shadow-primary/5 sm:gap-4 sm:p-4"
   469→    >
   470→      {/* Category color left accent */}
   471→      <span
   472→        className={`absolute inset-y-0 left-0 w-1 bg-gradient-to-b ${gradient} opacity-60 transition-opacity group-hover:opacity-100`}
   473→        aria-hidden
   474→      />
   475→      {/* Decorative corner wash */}
   476→      <span
   477→        className={`pointer-events-none absolute -right-10 -top-10 h-24 w-24 rounded-full bg-gradient-to-br ${gradient} opacity-[0.07] blur-2xl transition group-hover:opacity-20`}
   478→        aria-hidden
   479→      />
   480→
   481→      {/* Stretched link: whole row stays clickable; sits ABOVE static content
   482→          (z-[1]) but BELOW interactive controls (details summary, actions). */}
   483→      <Link
   484→        href={`/grupo/${group.slug}`}
   485→        title={imgTitle}
   486→        aria-label={group.isAdult ? "Ver grupo 18+" : `Ver grupo ${group.title}`}
   487→        className="absolute inset-0 z-[1] rounded-2xl"
   488→      />

   594→
   595→function Highlight({ text, query }: { text: string; query: string }) {
   596→  if (!query || query.trim().length === 0) return <>{text}</>;
   597→  const escaped = query.trim().replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
   598→  const regex = new RegExp(`(${escaped})`, "gi");
   599→  const parts = text.split(regex);
   600→  return (
   601→    <>
   602→      {parts.map((part, i) =>
   603→        part.toLowerCase() === query.trim().toLowerCase() ? (
   604→          <mark key={i} className="rounded bg-primary/20 px-0.5 text-primary">{part}</mark>
   605→        ) : (
   606→          <React.Fragment key={i}>{part}</React.Fragment>
   607→        )
   608→      )}
   609→    </>
   610→  );
   611→}
