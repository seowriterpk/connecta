    44→    44→// Strict page parsing: absent param or junk (?page=abc → NaN) falls back to
    45→    45→// page 1; 0/negative/non-integer values pass through and are rejected by the
    46→    46→// paginated data function (out of range → notFound).
    47→    47→function parsePageParam(sp: { [key: string]: string | string[] | undefined }): number {
    48→    48→  const raw = Array.isArray(sp.page) ? sp.page[0] : sp.page;
    49→    49→  if (raw === undefined) return 1;
    50→    50→  const n = Number(raw);
    51→    51→  return Number.isFinite(n) ? n : 1;
    52→    52→}

    54→    54→export async function generateMetadata({
    55→    55→  params,
    56→    56→  searchParams,
    57→    57→}: PageProps): Promise<Metadata> {
    58→    58→  const { slug } = await params;
    59→    59→  const sp = await searchParams;
    60→    60→  const page = parsePageParam(sp);
    61→    61→  const city = await getCityBySlug(slug);
    62→    62→  if (!city) {
    63→    63→    return {
    64→    64→      title: "Ciudad no encontrada",
    65→    65→      robots: { index: false, follow: false },
    66→    66→    };
    67→    67→  }
    68→    68→
    69→    69→  const basePath = `/ciudad/${city.slug}`;
    70→    70→  const canonicalPath = page === 1 ? basePath : `${basePath}?page=${page}`;
    71→    71→  const canonical = `${SITE.url}${canonicalPath}`;
    72→    72→
    73→    73→  // Count-first CTR title — N is the same live groupCount the page header
    74→    74→  // renders (getAllCities-derived; reuse, no extra queries).
    75→    75→  const title = `${city.groupCount > 0 ? `${city.groupCount} ` : ""}Grupos de WhatsApp en ${city.city} | Grupos Hoy`;
    76→    76→  // D-4 thin-content rule: city pages with fewer than 3 groups are
    77→    77→  // template-swapped thin pages → keep them live for users but noindex.
    78→    78→  const robots =
    79→    79→    city.groupCount >= 3
    80→    80→      ? { index: true, follow: true }
    81→    81→      : { index: false, follow: true };
    82→    82→  // Dynamic meta description: custom hero override wins, else keyword-derived
    83→    83→  // from the oldest groups of this city (one keyword per group).
    84→    84→  const taxContent = await getTaxonomyContent("city", city.city);
    85→    85→  const description = buildTaxonomyMetaDescription(
    86→    86→    "city",
    87→    87→    city.city,
    88→    88→    taxContent,
    89→    89→    city.groupCount
    90→    90→  );
    91→    91→
    92→    92→  return {
    93→    93→    title: { absolute: title },
    94→    94→    description,
    95→    95→    alternates: { canonical },
    96→    96→    robots,
    97→    97→    openGraph: {
    98→    98→      title,
    99→    99→      description,
   100→   100→      url: canonical,
   101→   101→      type: "website",
   102→   102→      locale: "es_ES",
   103→   103→      siteName: SITE.name,
   104→   104→    },
   105→   105→    twitter: {
   106→   106→      card: "summary_large_image",
   107→   107→      title,
   108→   108→      description,
   109→   109→    },
   110→   110→    keywords: [
   111→   111→      `grupos de WhatsApp en ${city.city}`,
   112→   112→      `grupos de WhatsApp ${city.city}`,
   113→   113→      `${city.city} WhatsApp`,
   114→   114→      `grupos WhatsApp ${city.countryCode.toUpperCase()}`,
   115→   115→      "grupos de WhatsApp en español",
   116→   116→      "unirse a grupo de WhatsApp",
   117→   117→      "comunidades hispanohablantes",
   118→   118→      `enlaces de grupos de WhatsApp ${city.city}`,
   119→   119→    ],
   120→   120→  };
   121→   121→}
