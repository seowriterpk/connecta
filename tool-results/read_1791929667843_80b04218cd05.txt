   595→   595→function Highlight({ text, query }: { text: string; query: string }) {
   596→   596→  if (!query || query.trim().length === 0) return <>{text}</>;
   597→   597→  const escaped = query.trim().replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
   598→   598→  const regex = new RegExp(`(${escaped})`, "gi");
   599→   599→  const parts = text.split(regex);
   600→   600→  return (
   601→   601→    <>
   602→   602→      {parts.map((part, i) =>
   603→   603→        part.toLowerCase() === query.trim().toLowerCase() ? (
   604→   604→          <mark key={i} className="rounded bg-primary/20 px-0.5 text-primary">{part}</mark>
   605→   605→        ) : (
   606→   606→          <React.Fragment key={i}>{part}</React.Fragment>
   607→   607→        )
   608→   608→      )}
   609→   609→    </>
   610→   610→  );
   611→   611→}
