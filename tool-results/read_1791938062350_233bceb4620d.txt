   475→function GroupRow({ group, searchQuery }: { group: GroupCardDTO; searchQuery: string }) {
   476→  const isActiveRecently = group.lastActiveAt
   477→    ? Date.now() - new Date(group.lastActiveAt).getTime() < 7 * 86400000
   478→    : false;
   479→  const isHot = group.members >= 1000;
   480→  const isTrending = group.views >= 50 || group.members >= 700;
   481→  const gradient = ROW_GRADIENTS[group.category?.color ?? "emerald"] ?? ROW_GRADIENTS.emerald;
   482→  // Adult directive: blank image title/alt on adult groups (SEO safety).
   483→  const imgAlt = group.isAdult ? "" : group.title;
   484→  const imgTitle = group.isAdult ? undefined : group.title;
   485→
   486→  return (
   487→    <motion.article

   616→
   617→function Highlight({ text, query }: { text: string; query: string }) {
   618→  if (!query || query.trim().length === 0) return <>{text}</>;
   619→  const escaped = query.trim().replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
   620→  const regex = new RegExp(`(${escaped})`, "gi");
   621→  const parts = text.split(regex);
   622→  return (
   623→    <>
   624→      {parts.map((part, i) =>
   625→        part.toLowerCase() === query.trim().toLowerCase() ? (
   626→          <mark key={i} className="rounded bg-primary/20 px-0.5 text-primary">{part}</mark>
   627→        ) : (
   628→          <React.Fragment key={i}>{part}</React.Fragment>
   629→        )
   630→      )}
   631→    </>
   632→  );
   633→}
