    39→interface PageProps {
    40→  params: Promise<{ code: string }>;
    41→  searchParams: Promise<{ [key: string]: string | string[] | undefined }>;
    42→}
    43→
    44→// Strict page parsing: absent param or junk (?page=abc → NaN) falls back to
    45→// page 1; 0/negative/non-integer values pass through and are rejected by the
    46→// paginated data function (out of range → notFound).
    47→function parsePageParam(sp: { [key: string]: string | string[] | undefined }): number {
    48→  const raw = Array.isArray(sp.page) ? sp.page[0] : sp.page;
    49→  if (raw === undefined) return 1;
    50→  const n = Number(raw);
    51→  return Number.isFinite(n) ? n : 1;
    52→}

    54→export async function generateMetadata({
    55→  params,
    56→  searchParams,
    57→}: PageProps): Promise<Metadata> {
    58→  const { code } = await params;
    59→  const sp = await searchParams;
    60→  const page = parsePageParam(sp);
    61→  const country = await getCountryByCode(code);
    62→  if (!country) {
    63→    return {
    64→      title: "País no encontrado",
    65→      robots: { index: false, follow: false },
    66→    };
    67→  }
    68→
    69→  const basePath = `/pais/${country.code}`;
    70→  const canonicalPath = page === 1 ? basePath : `${basePath}?page=${page}`;
    71→  const canonical = `${SITE.url}${canonicalPath}`;
    72→
    73→  // Count-first CTR title — N is the same clean live groupCount the page
    74→  // header renders (getCountryByCode → getGroupsCount, clean by default).
    75→  const title = `${country.groupCount > 0 ? `${country.groupCount} ` : ""}Grupos de WhatsApp de ${country.name} | Grupos Hoy`;
    76→  // D-4 thin-content rule: country pages with fewer than 3 groups are
    77→  // template-swapped thin pages → keep them live for users but noindex.
    78→  const robots =
    79→    country.groupCount >= 3
    80→      ? { index: true, follow: true }
    81→      : { index: false, follow: true };
    82→  // Dynamic meta description: custom hero override wins, else keyword-derived
    83→  // from the oldest groups of this country (one keyword per group).
    84→  const taxContent = await getTaxonomyContent("country", country.name);
    85→  const description = buildTaxonomyMetaDescription(
    86→    "country",
    87→    country.name,
    88→    taxContent,
    89→    country.groupCount
    90→  );
    91→
    92→  return {
    93→    title: { absolute: title },
    94→    description,
    95→    alternates: { canonical },
    96→    robots,
    97→    openGraph: {
    98→      title,
    99→      description,
   100→      url: canonical,
   101→      type: "website",
   102→      locale: "es_ES",
   103→      siteName: SITE.name,
   104→    },
   105→    twitter: {
   106→      card: "summary_large_image",
   107→      title,
   108→      description,
   109→    },
   110→    keywords: [
   111→      `grupos de WhatsApp en ${country.name}`,
   112→      `grupos de WhatsApp ${country.code.toUpperCase()}`,
   113→      `${country.name} WhatsApp`,
   114→      "grupos de WhatsApp en español",
   115→      "unirse a grupo de WhatsApp",
   116→      "comunidades hispanohablantes",
   117→      `enlaces de grupos de WhatsApp ${country.name}`,
   118→      country.region,
   119→    ],
   120→  };
   121→}
