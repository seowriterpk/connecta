    39→interface PageProps {

    43→
    44→// Strict page parsing: absent param or junk (?page=abc → NaN) falls back to
    45→// page 1; 0/negative/non-integer values pass through and are rejected by the
    46→// paginated data function (out of range → notFound).
    47→function parsePageParam(sp: { [key: string]: string | string[] | undefined }): number {
    48→  const raw = Array.isArray(sp.page) ? sp.page[0] : sp.page;
    49→  if (raw === undefined) return 1;
    50→  const n = Number(raw);
    51→  return Number.isFinite(n) ? n : 1;
    52→}
    53→
    54→export async function generateMetadata({
    55→  params,
    56→  searchParams,
    57→}: PageProps): Promise<Metadata> {
    58→  const { slug } = await params;
    59→  const sp = await searchParams;
    60→  const page = parsePageParam(sp);
    61→  const city = await getCityBySlug(slug);
    62→  if (!city) {
    63→    return {
    64→      title: "Ciudad no encontrada",
    65→      robots: { index: false, follow: false },
    66→    };
    67→  }
    68→
    69→  const basePath = `/ciudad/${city.slug}`;
    70→  const canonicalPath = page === 1 ? basePath : `${basePath}?page=${page}`;
    71→  const canonical = `${SITE.url}${canonicalPath}`;
    72→
    73→  // Count-first CTR title — N is the same live groupCount the page header
    74→  // renders (getAllCities-derived; reuse, no extra queries).
    75→  const title = `${city.groupCount > 0 ? `${city.groupCount} ` : ""}Grupos de WhatsApp en ${city.city} | Grupos Hoy`;
    76→  // D-4 thin-content rule: city pages with fewer than 3 groups are
    77→  // template-swapped thin pages → keep them live for users but noindex.
    78→  const robots =
    79→    city.groupCount >= 3
    80→      ? { index: true, follow: true }
    81→      : { index: false, follow: true };
    82→  // Dynamic meta description: custom hero override wins, else keyword-derived
    83→  // from the oldest groups of this city (one keyword per group).
    84→  const taxContent = await getTaxonomyContent("city", city.city);
    85→  const description = buildTaxonomyMetaDescription(
    86→    "city",
    87→    city.city,
    88→    taxContent,
    89→    city.groupCount
    90→  );
    91→
    92→  return {
    93→    title: { absolute: title },
    94→    description,
    95→    alternates: { canonical },
    96→    robots,
    97→    openGraph: {
    98→      title,
    99→      description,
   100→      url: canonical,
   101→      type: "website",
   102→      locale: "es_ES",
   103→      siteName: SITE.name,
   104→    },
   105→    twitter: {
   106→      card: "summary_large_image",
   107→      title,
   108→      description,
   109→    },
   110→    keywords: [
   111→      `grupos de WhatsApp en ${city.city}`,
   112→      `grupos de WhatsApp ${city.city}`,
   113→      `${city.city} WhatsApp`,
   114→      `grupos WhatsApp ${city.countryCode.toUpperCase()}`,
   115→      "grupos de WhatsApp en español",
   116→      "unirse a grupo de WhatsApp",
   117→      "comunidades hispanohablantes",
   118→      `enlaces de grupos de WhatsApp ${city.city}`,
   119→    ],
   120→  };
   121→}
